/**
 * Screenshot harness. Everything here is hardcoded on purpose:
 *
 * - no file access, no permissions, no config: the theme comes from the launch intent only;
 * - every string is a pre-generated random token rendered with the Sarcaz font, and the very same list
 *   is used for the light and the dark capture so the pair differs in colour alone;
 * - the system bars are hidden, so the top row of pixels is the top of the title bar - that is where
 *   the 3:2 crop starts.
 */
class MainActivity : ComponentActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val dark = intent.getIntExtra(EXTRA_THEME, THEME_LIGHT) == THEME_DARK
        WindowCompat.setDecorFitsSystemWindows(window, false)
        WindowInsetsControllerCompat(window, window.decorView).apply {
            hide(WindowInsetsCompat.Type.systemBars())
            systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        }
        setContent { ShotsTheme(dark) { MockWorlds() } }
    }

    companion object {
        const val EXTRA_THEME = "theme"
        const val THEME_LIGHT = 0
        const val THEME_DARK = 1
    }
}

private val Sarcaz = FontFamily(Font(R.font.endfield_by_butan))

private fun sarcazTypography(): Typography {
    val base = Typography()
    fun TextStyle.withSarcaz() = copy(fontFamily = Sarcaz)
    return Typography(
        displayLarge = base.displayLarge.withSarcaz(),
        displayMedium = base.displayMedium.withSarcaz(),
        displaySmall = base.displaySmall.withSarcaz(),
        headlineLarge = base.headlineLarge.withSarcaz(),
        headlineMedium = base.headlineMedium.withSarcaz(),
        headlineSmall = base.headlineSmall.withSarcaz(),
        titleLarge = base.titleLarge.withSarcaz(),
        titleMedium = base.titleMedium.withSarcaz(),
        titleSmall = base.titleSmall.withSarcaz(),
        bodyLarge = base.bodyLarge.withSarcaz(),
        bodyMedium = base.bodyMedium.withSarcaz(),
        bodySmall = base.bodySmall.withSarcaz(),
        labelLarge = base.labelLarge.withSarcaz(),
        labelMedium = base.labelMedium.withSarcaz(),
        labelSmall = base.labelSmall.withSarcaz(),
    )
}

@Composable
private fun ShotsTheme(dark: Boolean, content: @Composable () -> Unit) {
    MaterialTheme(
        colorScheme = if (dark) ShotsDark else ShotsLight,
        typography = sarcazTypography(),
        content = content,
    )
}

// ---------------------------------------------------------------------------------------------
// Layout: the same shapes as the world list page, nothing interactive.
// ---------------------------------------------------------------------------------------------

@Composable
private fun MockWorlds() {
    Surface(Modifier.fillMaxSize(), color = MaterialTheme.colorScheme.background) {
        Column(Modifier.fillMaxSize().verticalScroll(rememberScrollState())) {
            TitleBar()
            Column(Modifier.padding(horizontal = 16.dp)) {
                Spacer(Modifier.height(8.dp))
                RootChips()
                Spacer(Modifier.height(12.dp))
                NoticeCard()
                Spacer(Modifier.height(12.dp))
                Worlds.forEach { world ->
                    WorldCard(world.first, world.second, world.third)
                    Spacer(Modifier.height(12.dp))
                }
                FooterBar()
            }
        }
    }
}

@Composable
private fun TitleBar() {
    Row(
        Modifier.fillMaxWidth().padding(start = 16.dp, end = 16.dp, top = 16.dp, bottom = 8.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(
            text = Title,
            style = MaterialTheme.typography.headlineSmall,
            color = MaterialTheme.colorScheme.onBackground,
            modifier = Modifier.weight(1f),
        )
        Icon(
            Icons.Outlined.Restore,
            contentDescription = null,
            tint = MaterialTheme.colorScheme.onBackground,
            modifier = Modifier.size(26.dp),
        )
        Spacer(Modifier.width(20.dp))
        Icon(
            Icons.Outlined.Refresh,
            contentDescription = null,
            tint = MaterialTheme.colorScheme.onBackground,
            modifier = Modifier.size(26.dp),
        )
    }
}

@Composable
private fun RootChips() {
    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        Roots.forEachIndexed { index, label ->
            val selected = index == 0
            Surface(
                shape = RoundedCornerShape(10.dp),
                color = if (selected) {
                    MaterialTheme.colorScheme.secondaryContainer
                } else {
                    Color.Transparent
                },
                border = if (selected) null else BorderStroke(1.dp, MaterialTheme.colorScheme.outline),
            ) {
                Text(
                    text = label,
                    style = MaterialTheme.typography.labelLarge,
                    color = if (selected) {
                        MaterialTheme.colorScheme.onSecondaryContainer
                    } else {
                        MaterialTheme.colorScheme.onSurfaceVariant
                    },
                    modifier = Modifier.padding(horizontal = 18.dp, vertical = 12.dp),
                )
            }
        }
    }
}

@Composable
private fun NoticeCard() {
    Surface(
        Modifier.fillMaxWidth(),
        shape = RoundedCornerShape(16.dp),
        color = MaterialTheme.colorScheme.surfaceVariant,
    ) {
        Text(
            text = Notice,
            style = MaterialTheme.typography.bodyMedium,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
            modifier = Modifier.padding(20.dp),
        )
    }
}

@Composable
private fun WorldCard(title: String, chips: List<String>, note: String) {
    Surface(
        Modifier.fillMaxWidth(),
        shape = RoundedCornerShape(16.dp),
        color = MaterialTheme.colorScheme.surfaceVariant,
    ) {
        Column(Modifier.padding(20.dp)) {
            Row(verticalAlignment = Alignment.Top) {
                Icon(
                    Icons.Outlined.Map,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.primary,
                    modifier = Modifier.size(28.dp),
                )
                Spacer(Modifier.width(14.dp))
                Text(
                    text = title,
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                    modifier = Modifier.weight(1f),
                )
                Checkbox(checked = false, onCheckedChange = null)
            }
            Spacer(Modifier.height(12.dp))
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                chips.forEach { label ->
                    Surface(
                        shape = RoundedCornerShape(8.dp),
                        color = Color.Transparent,
                        border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant),
                    ) {
                        Text(
                            text = label,
                            style = MaterialTheme.typography.labelMedium,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.padding(horizontal = 12.dp, vertical = 8.dp),
                        )
                    }
                }
            }
            Spacer(Modifier.height(10.dp))
            Text(
                text = note,
                style = MaterialTheme.typography.bodySmall.copy(fontSize = 13.sp),
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }
    }
}

@Composable
private fun FooterBar() {
    Row(
        Modifier.fillMaxWidth().padding(vertical = 16.dp),
        horizontalArrangement = Arrangement.spacedBy(24.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(
            text = Footer[0],
            style = MaterialTheme.typography.labelLarge,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
        )
        Text(
            text = Footer[1],
            style = MaterialTheme.typography.labelLarge,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
        )
    }
}
